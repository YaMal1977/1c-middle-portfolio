Процедура УстановитьДопСведенияДокументаЗаказКлиента()
	// Свойства и их значения
	
	Запрос = Новый Запрос;
	Запрос.Текст = "ВЫБРАТЬ
	               |	ЗаказыКлиентов.ЗаказКлиентаПоставщика КАК ЗаказКлиента,
	               |	ЗаказыКлиентов.Ссылка.ВидОперации КАК ВидОперации,
	               |	ЗаказыКлиентов.Ссылка.Статус КАК Статус,
	               |	ЗаказыКлиентов.ОтветственныйЗаСтатус КАК ОтветственныйЗаСтатус
	               |ИЗ
	               |	Документ.ЭК_ПутевойЛист.ЗаказыКлиентов КАК ЗаказыКлиентов
	               |ГДЕ
	               |	ЗаказыКлиентов.Ссылка.ПометкаУдаления = ЛОЖЬ
	               |	И ЗаказыКлиентов.Ссылка = &Ссылка";
	
	Запрос.УстановитьПараметр("Ссылка", Ссылка); // СсылкаНаПутевойЛист - это ссылка на документ ЭК_ПутевойЛист
	
	Результат = Запрос.Выполнить();
	
	Если Не Результат.Пустой() Тогда
		Выборка = Результат.Выбрать();
		Пока Выборка.Следующий() Цикл
			ЗаказКлиентаСсылка = Выборка.ЗаказКлиента;
			
			СвойстваИЗначения = Новый Соответствие;
			
			///--Свойств может быть несколько
			
			СвойстваИЗначения.Вставить("Свойство1", "Значение1");
			///СвойстваИЗначения.Вставить("Свойство2", "Значение2");
			
			Для Каждого Элемент Из СвойстваИЗначения Цикл
				УстановитьДополнительноеСведение(ЗаказКлиентаСсылка, Элемент.Ключ, Элемент.Значение);
			КонецЦикла;
			
		КонецЦикла;
	КонецЕсли;

//  устанавливаем дополнительные сведения
Процедура УстановитьДополнительноеСведение(ОбъектСсылка, НаименованиеСвойства, Значение)
	Свойство = ПланыВидовХарактеристик.ДополнительныеРеквизитыИСведения.НайтиПоНаименованию(НаименованиеСвойства);
	Если Свойство <> Неопределено И Свойство.ЭтоДополнительноеСведение Тогда
		МенеджерЗаписи = РегистрыСведений.ДополнительныеСведения.СоздатьМенеджерЗаписи();
		МенеджерЗаписи.Объект = ОбъектСсылка;
		МенеджерЗаписи.Свойство = Свойство;
		МенеджерЗаписи.Прочитать();
		Если МенеджерЗаписи.Выбран() Тогда
			МенеджерЗаписи.Значение = Значение;
		Иначе
			МенеджерЗаписи.Значение = Значение;
			МенеджерЗаписи.Записать(Истина);
		КонецЕсли;
	КонецЕсли;
КонецПроцедуры
