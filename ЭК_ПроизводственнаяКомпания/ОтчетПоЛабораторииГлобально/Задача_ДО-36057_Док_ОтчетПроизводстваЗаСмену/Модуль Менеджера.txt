	ИначеЕсли УправлениеПечатью.НужноПечататьМакет(КоллекцияПечатныхФорм, "ЛистКонтроля") Тогда

			,
			"Документ.ЭК_ОтчетПроизводстваЗаСмену.ЭК_ЛистКонтроля");
    ИначеЕсли УправлениеПечатью.НужноПечататьМакет(КоллекцияПечатныхФорм, "ЭК_СоставРабочих") Тогда

			КоллекцияПечатныхФорм,
			"ЭК_СоставРабочих",

			СформироватьПечатнуюФормуСоставРабочих(МассивОбъектов, ОбъектыПечати, ПараметрыПечати),
			,
			"Документ.ЭК_ОтчетПроизводстваЗаСмену.ЭК_СоставРабочих");

			СтрЧ.КалендарныйПериод = СтрЧ.ДлительностьВыпуска / (КоличествоДней) * 100;

				ЧасыПростоя = СтрЧ.ПростойОпер + СтрЧ.МойкаПлановая + СтрЧ.ЗаменаРасходныхМатериалов + СтрЧ.ПростойСТС + СтрЧ.ПростойСПО + СтрЧ.КупажноеОтделение + СтрЧ.Лаборатория + СтрЧ.ФорсМажор + СтрЧ.ПереходКупажа;

Функция ЧасыВСтроку(А) Экспорт
	Если А <= 0 Тогда
		Возврат "";
	КонецЕсли;	
	Часы = Цел(А / 3600);
	Минуты = Окр((А - (Часы * 3600))/60);
	Если Минуты < 1 Тогда 
		Минуты = 0;	
	КонецЕсли;
	Если Минуты = 60 Тогда 
		Часы = Часы + 1;
		Минуты = 0;
	КонецЕсли;
			
	Стр = Формат(Часы, "ЧН=; ЧГ=0");
	Если СтрДлина(Стр) < 2 Тогда
		Стр = "0" + Стр;
	КонецЕсли;
	Стр = Стр + ":" + Прав("0" + Формат(Минуты, "ЧН=; ЧГ=0"), 2);
	Возврат Стр;

Функция ПолучитьЧасыРаботы(Количество, Производительность)
	ЧасыРаботы = Количество / Производительность * 3600;
	ЧасРабота = Цел(ЧасыРаботы);
	МинутаРабота = (ЧасыРаботы - ЧасРабота);
	
	Возврат ЧасРабота + Цел(МинутаРабота);

Функция ПриведемСтрокуКСтандарту(ТекСтрока)
	Знак = СтрНайти(ТекСтрока,",") - 1;
	Если Знак = 0 Тогда
		Часы = Число(ТекСтрока);
		Минуты = "0";
	Иначе
		Часы = Число(Лев(ТекСтрока,Знак));
		Минуты = Число(Прав(ТекСтрока,Знак));
	КонецЕсли;
	
	Если Часы < 10 Тогда 
		Часы = "0" + Строка(Часы);	
	Иначе
		Часы = Строка(Часы);	
	КонецЕсли;
	
	Если Минуты < 10 Тогда 
		Минуты = Минуты / 10 * 60;
	Иначе
		Минуты = Минуты / 100 * 60;	
	КонецЕсли;
	Если Минуты < 10 Тогда 
		Минуты = "0" + Строка(Минуты);	
	Иначе
		Минуты = Строка(Минуты);	
	КонецЕсли;
	
	Возврат Часы + ":" + Минуты;

Функция СтрокаВЧисло(ТекСтрока)
	Если ТекСтрока = "" Тогда 
		Часы = 0;
		Минуты = 0;
	Иначе
		ДлинаСтроки = СтрДлина(ТекСтрока);
		Знак = СтрНайти(ТекСтрока,":") - 1;
		Часы = Число(Лев(ТекСтрока,Знак)) * 60 * 60;
		Минуты = Число(Прав(ТекСтрока,ДлинаСтроки - Знак - 1))*60;
	КонецЕсли;
	Возврат Часы + Минуты

Функция ЧислоВДату(А)
	Часов = Цел(А / 3600);
	Минут = Цел((А / 3600 - Часов) * 60);
	Секунд = Окр(((А / 3600 - Часов) * 60 - Минут) * 60, 0);
	Если Секунд = 60 Тогда 
		Секунд = 0;
		Минут = Минут+1;
	КонецЕсли;
	Если Минут = 60 Тогда 
		Минут = 0;
		Часов = Часов + 1;
	КонецЕсли;
	Попытка
		ВремяДляПоказа = Дата(1, 1, 1, Часов, Минут, Секунд);
	Исключение 
		ВремяДляПоказа = Дата(1, 1, 1, 00, 00, 1);	
	КонецПопытки;
	Возврат ВремяДляПоказа;
